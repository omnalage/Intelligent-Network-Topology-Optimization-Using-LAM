\section{Introduction}
\begin{frame}
  \frametitle{Introduction to ICN Architecture}
  \begin{itemize}
    \item<1-> Shift from Host-Centric (IP) to Content-Centric (focus on content, not location).
    \item<2-> Routers act as forwarders and in-network caches (dual role).
    \item<3-> \textbf{Packet Types:} Interest (request) and Data (respons
    with content).
    \item<4-> \textbf{Router Data Structures:}
    \begin{itemize}
        \item Content Store (CS)
        \item Pending Interest Table (PIT)
        \item Forwarding Info Base (FIB)
    \end{itemize}
    \item<5-> \textbf{Communication Flow:} Client sends an Interest, routers forward/cache, the server/cache responds with Data, and the Data packet returns along the path, being cached at intermediate routers.
  \end{itemize}
\end{frame}

\section{Background}
\begin{frame}
  \frametitle{Background Concepts}
  \begin{itemize}
    \item \textbf{Content-Centric Networking (CCN):} Focuses on content availability and retrieval, not the host's location.
    \item \textbf{Centrality Measures:} Metrics to identify the most important or "central" routers in a network topology.
    \item \textbf{CMBA (Centrality-Measures Based Algorithm):} Uses centrality measures to select the most appropriate routers for caching content to improve performance.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Centrality Measure Formulas}
  \justify
  Here are the core formulas used to identify influential nodes in the network.
  \begin{columns}[T]
    % Column 1: Degree and Closeness
    \begin{column}{0.5\textwidth}
      \begin{block}{Degree Centrality $C_D(v)$}
      The number of direct connections a node has.
      \begin{equation}
DC(CR_i) = \frac{degree(CR_i)}{(n - 1)}
\end{equation}

      \end{block}
      \begin{block}{Closeness Centrality $C_C(v)$}
      The reciprocal of the sum of the shortest path distances to all other nodes.
      \begin{equation}
CC(CR_i) = \frac{(n - 1)}{\displaystyle \sum_{\substack{j = 1 \\ j \neq i}}^{n} d(CR_i, CR_j)}
\end{equation}

      \end{block}
    \end{column}
     \end{columns}
   \end{frame}
   
    \begin{frame}
    
     \begin{columns}
        
    
    % Column 2: Betweenness and Reachness
    \begin{column}{0.5\textwidth}
      \begin{block}{Betweenness Centrality $C_B(v)$}
      The fraction of all-pairs shortest paths that pass through a node.
      \begin{equation}
          BC(CR_i) = \sum_{i \neq CR_i \neq j \in n} \frac{\sigma_{i,j}(CR_i)}{\sigma_{i,j}}
      \end{equation}
      \end{block}
      \begin{block}{Reach Centrality $C_R(v)$}
      It defines how many no. of hops, a particular CR reaches to another CR
     \begin{equation}
RC(CR_i) = 1 + \sum_{x = 1}^{e(CR_i)} \frac{r_{ix}}{x}
\end{equation}

      
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

%Algorithm portion
\begin{frame}[fragile]
    \frametitle{Algorithm: Content Request}
    \begin{block}{Algorithm 1: Content request}
    \begin{algorithmic}[1]
        \State Initialize (CMV=0)
        
        \For{for each ($CR_n$ from i to j do )}
        
            \If{if content item in the CS of CR then }
            
                \State send towards the client.
                
                \State send(data)
                
            \Else else
            
                \State      Calculate CMV at each CR upon arriving of interest with respect to a client.
                
                \If{if calculated value of CMV $>$ CMV stored in an interest}
                
                    \State Then, update CMV with router-id into interest.
                    
                \Else else
                
                    \State Forward request to the next hop towards j.
                \EndIf
            \EndIf
        \EndFor
    \end{algorithmic}
    \end{block}
\end{frame}
